cmake_minimum_required(VERSION 4.4.5)
if(CMAKE_GENERATOR MATCHES "^Visual Studio" AND NOT CMAKE_GENERATOR STREQUAL "Visual Studio 18 2026")
  message(FATAL_ERROR "Ursa requires Visual Studio 2026 (v145). Use a new build directory with -G \"Visual Studio 18 2026\" -A x64 -T v145; do not reuse a VS2022 cache.")
endif()
project(UrsaEngine VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
# Keep the debug library naming consistent before dependencies configure their own cache settings.
set(CMAKE_DEBUG_POSTFIX "d")

set(URSA_SOURCE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}")
set(URSA_BINARY_ROOT "${CMAKE_CURRENT_BINARY_DIR}")
set_property(GLOBAL PROPERTY USE_FOLDERS ON)
set(URSA_PROJECT_DIR "" CACHE PATH "Optional C# project path used by Engine integration tests and package commands")
if(NOT URSA_PROJECT_DIR STREQUAL "")
  get_filename_component(URSA_PROJECT_DIR "${URSA_PROJECT_DIR}" ABSOLUTE)
  if(NOT EXISTS "${URSA_PROJECT_DIR}/Assets")
    message(FATAL_ERROR "URSA_PROJECT_DIR must point to a project with an Assets directory: ${URSA_PROJECT_DIR}")
  endif()
endif()


option(URSA_BUILD_EDITOR "Build the Ursa editor" ON)
option(URSA_ENABLE_DOTNET "Enable the optional .NET 10 scripting host" OFF)
option(URSA_ENABLE_TRACY "Enable optional Tracy CPU and Vulkan GPU profiling" OFF)
option(URSA_ENABLE_TRACY_MEMORY "Trace Ursa CPU and Vulkan device-block memory allocations" OFF)
option(URSA_ENABLE_TRACY_DIAGNOSTICS "Trace diagnostic plots, application events and selected locks" OFF)
set(URSA_TRACY_MEMORY_CALLSTACK_DEPTH "0" CACHE STRING "Native allocation callstack depth (0-64)")
if((URSA_ENABLE_TRACY_MEMORY OR URSA_ENABLE_TRACY_DIAGNOSTICS) AND NOT URSA_ENABLE_TRACY)
  message(FATAL_ERROR "Tracy memory and diagnostics require URSA_ENABLE_TRACY=ON")
endif()
if(NOT URSA_TRACY_MEMORY_CALLSTACK_DEPTH MATCHES "^(0|[1-9][0-9]?)$"
   OR URSA_TRACY_MEMORY_CALLSTACK_DEPTH GREATER 64)
  message(FATAL_ERROR "URSA_TRACY_MEMORY_CALLSTACK_DEPTH must be an integer from 0 to 64")
endif()
option(URSA_WARNINGS_AS_ERRORS "Treat warnings in Ursa native and managed targets as errors" ON)


if(CMAKE_SOURCE_DIR STREQUAL PROJECT_SOURCE_DIR)
  set(URSA_BUILD_TESTS_DEFAULT ON)
else()
  set(URSA_BUILD_TESTS_DEFAULT OFF)
endif()
option(URSA_BUILD_TESTS "Build the engine tests and benchmarks" ${URSA_BUILD_TESTS_DEFAULT})
option(URSA_WORKSPACE_BOOTSTRAP "Configure native tools before preparing project IDE files" OFF)
mark_as_advanced(URSA_WORKSPACE_BOOTSTRAP)

include(CMake/CompilerOptions.cmake)
include(CMake/SourceGroups.cmake)
include(CMake/DotNet.cmake)
include(CMake/Dependencies.cmake)

if(MSVC)
  add_compile_options($<$<COMPILE_LANGUAGE:CXX>:/utf-8>)
endif()
find_package(Vulkan 1.2 REQUIRED)
find_program(URSA_GLSLC_EXECUTABLE NAMES glslc HINTS "$ENV{VULKAN_SDK}/Bin")
find_program(URSA_DXC_EXECUTABLE NAMES dxc HINTS "$ENV{VULKAN_SDK}/Bin")
find_program(URSA_SPIRV_VAL_EXECUTABLE NAMES spirv-val HINTS "$ENV{VULKAN_SDK}/Bin")
find_program(URSA_SPIRV_CROSS_EXECUTABLE NAMES spirv-cross HINTS "$ENV{VULKAN_SDK}/Bin")
foreach(tool IN ITEMS URSA_DXC_EXECUTABLE URSA_SPIRV_VAL_EXECUTABLE URSA_SPIRV_CROSS_EXECUTABLE)
  if(NOT ${tool})
    set(${tool} "")
  endif()
endforeach()
include(CMake/BuildResourcePaths.cmake)
ursa_configure_resource_paths()

add_subdirectory(Runtime)
add_subdirectory(Engine)
if(URSA_ENABLE_DOTNET)
  add_subdirectory(Managed)
  add_subdirectory(Engine/Scripting/DotNet)
endif()
if(URSA_BUILD_EDITOR)
  add_subdirectory(Editor)
endif()
add_subdirectory(Tools)
if(URSA_BUILD_TESTS)
  enable_testing()
  add_subdirectory(Tests)
endif()
if(URSA_WARNINGS_AS_ERRORS)
  ursa_configure_warning_policy("${URSA_SOURCE_ROOT}")
endif()
ursa_write_project_manifest()
include(CMake/IdeWorkspace.cmake)
ursa_configure_ide_workspace()
